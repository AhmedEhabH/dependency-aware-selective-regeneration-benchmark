    @classmethod
    def bulk_action(cls, info, queryset):
        sales = list(queryset)
        sale_ids = {sale.id for sale in sales}
        queryset.delete()
        for sale in sales:
            # Ensure we don't call plugin methods for sales that were already deleted
            if sale.id in sale_ids:
                previous_catalogue = fetch_catalogue_info(sale)
                info.context.plugins.sale_deleted(
                    sale, convert_catalogue_info_to_global_ids(previous_catalogue)
                )
