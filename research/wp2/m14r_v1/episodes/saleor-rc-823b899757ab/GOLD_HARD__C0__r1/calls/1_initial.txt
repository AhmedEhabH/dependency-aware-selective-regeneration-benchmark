class ProductSetAvailabilityForPurchase(BaseMutation):
    product = graphene.Field(Product)

    class Arguments:
        product_id = graphene.ID(
            required=True,
            description="ID of the product to set availability for purchase.",
        )
        is_available_for_purchase = graphene.Boolean(
            required=True,
            description="Determine if the product is available for purchase.",
        )
        start_date = graphene.Date(
            description="Start date of the availability for purchase. ISO 8601 standard."
        )

    class Meta:
        description = "Set product availability for purchase."
        permissions = (ProductPermissions.MANAGE_PRODUCTS,)
        error_type_class = ProductError
        error_type_field = "product_errors"

    @classmethod
    def perform_mutation(
        cls, _root, info, product_id, is_available_for_purchase, start_date=None
    ):
        product = cls.get_node_or_error(
            info, product_id, field="productId", only_type=Product
        )
        
        product.is_available_for_purchase = is_available_for_purchase
        
        if start_date:
            product.available_for_purchase = start_date
            
        product.save(
            update_fields=[
                "is_available_for_purchase", 
                "available_for_purchase"
            ]
        )
        
        return ProductSetAvailabilityForPurchase(product=product)
