def resolve_shipping_methods_for_checkout(
    info: ResolveInfo,
    checkout,
    manager,
    database_connection_name: str = settings.DATABASE_CONNECTION_DEFAULT_NAME,
):
    # Check if we're in a sync webhook context to prevent circular payload generation
    if hasattr(info.context, 'sync_webhook_control_context') and info.context.sync_webhook_control_context:
        # Return early to prevent circular references during payload generation
        return []
