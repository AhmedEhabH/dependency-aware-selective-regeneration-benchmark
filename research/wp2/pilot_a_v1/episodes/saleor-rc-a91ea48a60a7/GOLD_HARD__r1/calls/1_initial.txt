def validate_max_price(amount: Optional["Decimal"], currency: str):
    """Validate if price amount is not exceeding maximum allowed value."""

    # check no needed when there is no value
    if not amount:
        return

    currency_fraction = get_currency_fraction(currency)
    # Maximum price is typically limited by the currency's minor unit capacity
    # For most currencies, this would be 999,999,999,999.99 (12 digits before decimal)
    max_digits_before_decimal = 12
    max_allowed = 10 ** max_digits_before_decimal - (10 ** (-currency_fraction)) if currency_fraction > 0 else 10 ** max_digits_before_decimal
    
    if amount > max_allowed:
        raise ValidationError(
            f"Amount must be less than or equal to {max_allowed}."
        )
