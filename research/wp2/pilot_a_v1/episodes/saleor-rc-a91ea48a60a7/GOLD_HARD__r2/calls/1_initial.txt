    """Validate if price amount does not exceed the maximum value for the currency."""

    # check no needed when there is no value
    if not value:
        return
    from django_prices.utils.formatting import get_currency_max_amount

    max_amount = get_currency_max_amount(currency)

        raise ValidationError(
            f"Value cannot be higher than {max_amount} for currency {currency}."
